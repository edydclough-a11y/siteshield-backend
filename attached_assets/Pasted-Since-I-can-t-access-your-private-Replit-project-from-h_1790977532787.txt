```python
from fastapi import FastAPI, Form, UploadFile, File, HTTPException
from pydantic import BaseModel
from typing import List, Optional
from datetime import datetime
import uuid, os, shutil, re

app = FastAPI(title="SiteShield Ingest API")
ingested_data = []
os.makedirs("uploads", exist_ok=True)

CATEGORY_RULES = {
    "safety": [r"\bsafety\b", r"\bhazard\b", r"\bscaffold\b", r"\bfall\b", r"\bcaution\b"],
    "defect": [r"\bcrack\b", r"\bdamage\b", r"\bdefect\b", r"\bbroken\b", r"\brepair\b"],
    "quality": [r"\bsmudge\b", r"\bfinish\b", r"\binspection\b", r"\bquality\b"],
    "progress": [r"\bcomplet\b", r"\bbrickwork\b", r"\binstall\b", r"\bconstruct\b"],
    "environmental": [r"\brain\b", r"\bwater\b", r"\bdrainage\b", r"\bweather\b"],
}

def classify_note(text):
    if not text: return "general"
    lower = text.lower()
    scores = {c: sum(1 for p in pats if re.search(p, lower)) for c, pats in CATEGORY_RULES.items()}
    scores = {k: v for k, v in scores.items() if v > 0}
    return max(scores, key=scores.get) if scores else "general"

class IngestResponse(BaseModel):
    id: str; user_id: str; project_id: str
    title: Optional[str]; content: Optional[str]; category: str
    source: Optional[str]; file_name: Optional[str]; file_path: Optional[str]
    metadata: Optional[dict]; timestamp: str; status: str

@app.post("/ingest", response_model=IngestResponse)
async def ingest(
    user_id: str = Form(...),
    project_id: str = Form(...),
    raw_text: Optional[str] = Form(None),
    title: Optional[str] = Form(None),
    source: Optional[str] = Form(None),
    file: Optional[UploadFile] = File(None),
):
    file_name = file_path = None
    if file:
        file_name = file.filename
        file_path = f"uploads/{uuid.uuid4()}{os.path.splitext(file_name)[1]}"
        with open(file_path, "wb") as f: shutil.copyfileobj(file.file, f)
    record = {
        "id": str(uuid.uuid4()), "user_id": user_id, "project_id": project_id,
        "title": title, "content": raw_text, "category": classify_note(raw_text),
        "source": source, "file_name": file_name, "file_path": file_path,
        "metadata": {"user_id": user_id, "project_id": project_id},
        "timestamp": datetime.utcnow().isoformat() + "Z", "status": "ingested",
    }
    ingested_data.append(record)
    return record

@app.get("/ingest", response_model=List[IngestResponse])
def list_ingested(): return ingested_data

@app.get("/ingest/{item_id}", response_model=IngestResponse)
def get_ingested(item_id: str):
    for r in ingested_data:
        if r["id"] == item_id: return r
    raise HTTPException(status_code=404, detail="Not found")

if __name__ == "__main__":
    import uvicorn
    uvicorn.run("main:app", host="0.0.0.0", port=int(os.environ.get("PORT", "8000")))
```
